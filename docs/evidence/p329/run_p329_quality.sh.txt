#!/bin/bash
set -euo pipefail
base=/data/relcfd/chayo/physmorph_v2
code=$base/work/p303/code_production_withdrawal1
out=$base/work/p303/p329_raw_coast1.json
cd "$code"
test "$(cat VERSION)" = 19e725ec97e8229cafc8a1680bdebc23d45995fe
test ! -e "$out"
test -f "$base/work/p303/p329_live1/result.json"
export PHYSMORPH_RUN_REPO="$code"
source scripts/ops/hyde06_env.sh
source scripts/ops/gpu_env.sh
export CUDA_VISIBLE_DEVICES=0
exec 9>"$base/maintenance/gpu_launch.lock"
flock -x 9
now=$(date +%s); last=$(cat "$base/maintenance/last_gpu_launch_epoch")
if ((now-last<50)); then
    delay=$((50-now+last))
    ((delay<=50)) || exit 75
    sleep "$delay"
fi
used=$(du -sb "$base" | cut -f1)
((used+1000000<100000000000)) || exit 76
date +%s > "$base/maintenance/last_gpu_launch_epoch"
flock -u 9
exec "$PY" scripts/ops/cuda_python.py "$base/work/p303/p329_archive_quality.py" \
    --code "$code" --run "$base/work/p303/p329_live1" --out "$out"
