def identity(path):
    path=Path(path); before=path.stat(); digest=hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda:stream.read(8<<20),b''): digest.update(block)
    after=path.stat()
    assert (before.st_size,before.st_mtime_ns,before.st_ino)==(after.st_size,after.st_mtime_ns,after.st_ino)
    return dict(bytes=after.st_size,mtime_ns=after.st_mtime_ns,inode=after.st_ino,sha256=digest.hexdigest())

def bind(path, expected=None):
    value=identity(path)
    if expected is not None: assert value==expected,('bound file changed',str(path))
    bindings[str(path)]=value; return value

def read_json(path):
    value=bind(path); raw=Path(path).read_bytes()
    assert hashlib.sha256(raw).hexdigest()==value['sha256']
    return json.loads(raw)

def check(name, passed, **details):
    checks.append(dict(name=name,passed=bool(passed),**details))
    assert passed,name

def near(name, actual, expected, rel=3e-11, atol=1e-13):
    actual,expected=float(actual),float(expected)
    check(name,math.isfinite(actual) and math.isfinite(expected) and
          math.isclose(actual,expected,rel_tol=rel,abs_tol=atol),actual=actual,expected=expected)

def allowance(x): return 32*EPS*max(abs(x),1e-12)

def finite_json(value):
    if isinstance(value,dict): return {k:finite_json(v) for k,v in value.items()}
    if isinstance(value,(list,tuple)): return [finite_json(v) for v in value]
    return None if isinstance(value,float) and not math.isfinite(value) else value

def decode(tree, archive, convert):
    if not isinstance(tree,dict): return tree
    kind=tree['type']
    if kind=='tensor':
        x=archive[tree['key']]
        assert list(x.shape)==tree['shape'] and str(x.dtype)==tree['dtype']
        return convert(x)
    if kind=='dict': return {k:decode(v,archive,convert) for k,v in tree['items'].items()}
    values=[decode(v,archive,convert) for v in tree['items']]
    assert kind in ('tuple','list')
    return tuple(values) if kind=='tuple' else values
