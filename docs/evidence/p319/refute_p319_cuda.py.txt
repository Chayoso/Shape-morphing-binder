"""Independent P319 archive audit, no MPM or renderer.

All stored masks/coverage counts are reduced on CUDA. Source geometry is
recomputed only at preregistered initial/middle/worst/final samples and paired
raw/promoted endpoints. At those samples all point projections, but only a
bounded deterministic target-ID subset's brute distances, are recomputed.
"""
import argparse
from datetime import datetime, timezone
import hashlib
import json
import math
from pathlib import Path
import time
import zipfile

import numpy as np
import torch


def require(ok, text):
    if not ok:
        raise ValueError(text)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 << 20), b''):
            h.update(block)
    return h.hexdigest()


def identity(path):
    s = Path(path).stat()
    return [s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns]


def load_json(path):
    b = Path(path).read_bytes()
    result = json.loads(b)
    json.dumps(result, allow_nan=False)
    return result, hashlib.sha256(b).hexdigest()


def read_frame(path, index):
    with zipfile.ZipFile(path) as z, z.open('frames.npy') as stream:
        version = np.lib.format.read_magic(stream)
        shape, fortran, dtype = np.lib.format._read_array_header(stream, version)
        require(not fortran and dtype == np.float32 and len(shape) == 3 and shape[-1] == 3,
                'Unsupported raw frame layout')
        require(0 <= index < shape[0], 'Raw frame out of bounds')
        size = math.prod(shape[1:])*dtype.itemsize
        stream.seek(stream.tell()+index*size)
        b = stream.read(size)
        require(len(b) == size, 'Short frame')
        return torch.as_tensor(np.frombuffer(b, dtype).reshape(shape[1:]).copy(), device='cuda:0')


def unpack_rows(packed, width):
    require(packed.dtype == torch.uint8 and packed.ndim == 2 and
            packed.shape[1] == (width+7)//8, 'Packed row layout')
    shifts = torch.arange(7, -1, -1, dtype=torch.uint8, device=packed.device)
    bits = ((packed[..., None] >> shifts) & 1).bool().reshape(len(packed), -1)
    require(not bool(bits[:, width:].any()), 'Nonzero packed padding')
    return bits[:, :width]


def chosen_samples(samples, resolutions):
    """Metadata only; select before inspecting any source geometry."""
    chosen = {0: ['initial']}
    archived = [r for r in samples if r['archive_frame'] is not None]
    for row, reason in ((archived[len(archived)//2], 'middle_accepted_archive'),
                        (archived[-1], 'final_accepted_archive')):
        chosen.setdefault(row['sample_index'], []).append(reason)
    for res in resolutions:
        row = max(samples, key=lambda r: max(r['views'][str(res)]['internal_hole_pixels']))
        chosen.setdefault(row['sample_index'], []).append('maximum_single_view_holes_'+str(res))
    pairs = {r['paired_archive_frame'] for r in samples if r['sample_index'] in chosen and
             r['kind'] in ('optimizer_raw_endpoint', 'promoted_endpoint')}
    # Always inspect both final endpoints, even if the maximum-hole state is early.
    pairs.add(archived[-1]['paired_archive_frame'])
    for row in samples:
        if row['paired_archive_frame'] in pairs and row['kind'] in ('optimizer_raw_endpoint', 'promoted_endpoint'):
            chosen.setdefault(row['sample_index'], []).append('same_time_raw_promoted_pair')
    return {k: sorted(set(v)) for k, v in sorted(chosen.items())}


def target_ids(covered):
    """At most 128 IDs: 96 spaced over material ordering +16 covered/+16 missing."""
    n = len(covered)
    ids = torch.linspace(0, n-1, min(96, n), dtype=torch.float64, device=covered.device).round().long()
    return torch.cat((ids, torch.nonzero(covered).flatten()[:16],
                      torch.nonzero(~covered).flatten()[:16])).unique(sorted=True)


def projected_counts(x, res, az, el, extent):
    # Direct independent Torch implementation of the specified FP32 3x3 metric.
    right = x.new_tensor([math.cos(az), 0., -math.sin(az)])
    up = x.new_tensor([-math.sin(el)*math.sin(az), math.cos(el), -math.sin(el)*math.cos(az)])
    uv = torch.stack((x @ right, x @ up), 1)
    ij = torch.floor((uv+extent)/(2*extent)*res).long()
    valid = ((ij >= 0) & (ij < res)).all(1)
    selected = ij[valid]
    flat = torch.zeros(res*res, dtype=torch.int64, device=x.device)
    for ox in (-1, 0, 1):
        for oy in (-1, 0, 1):
            indices = (selected[:, 0]+ox).clamp(0, res-1)*res+(selected[:, 1]+oy).clamp(0, res-1)
            flat.scatter_add_(0, indices, torch.ones_like(indices))
    return flat.reshape(res, res), int((~valid).sum().item())


def main(folder, output):
    started = time.perf_counter()
    require(torch.cuda.is_available(), 'CUDA required')
    torch.cuda.set_device(0)
    torch.backends.cuda.matmul.allow_tf32 = False
    import cupy as cp
    from cupyx.scipy import ndimage
    from cupyx.scipy.spatial import KDTree
    require(int(cp.__version__.split('.')[0]) >= 14, 'CuPy14 exact device KDTree required')
    root = Path('/data/relcfd/chayo/physmorph_v2').resolve()
    folder, output = folder.resolve(), output.resolve()
    require(folder.is_relative_to(root) and output.is_relative_to(root), 'Outside /data project')
    require(not output.exists(), 'Exclusive output required')
    result, result_sha = load_json(folder/'result.json')
    samples = result['samples']
    require([r['sample_index'] for r in samples] == list(range(len(samples))), 'Sample index order')
    require(result['resolutions'] == [128, 256] and len(result['views']) == 24, 'Unexpected view/raster scope')
    d = result['discretization']
    require(d['N'] == 300000 and d['T'] == 20 and d['dt'] == 1/240, 'Unexpected discretization')
    paths = {**result['bindings'], **result['code'], str(folder/'result.json'): result_sha,
             str(folder/'target_masks.npz'): result['target_masks_sha256'],
             str(Path(__file__).resolve()): sha(__file__)}
    for chunk in result['chunks']:
        p = folder/chunk['path']
        require(p.resolve().is_relative_to(folder), 'Chunk path escape')
        paths[str(p)] = chunk['sha256']
    before = {p: identity(p) for p in paths}
    for p, h in paths.items():
        require(sha(p) == h, 'Hash mismatch: '+p)
    raw_paths = [Path(p) for p in result['bindings'] if p.endswith('_render_full_dt_iso_nn.npz')]
    trace_paths = [Path(p) for p in result['bindings'] if p.endswith('.rest_trace.json')]
    require(len(raw_paths) == len(trace_paths) == 1, 'Unique physical archive and trace')
    raw = raw_paths[0]
    trace, _ = load_json(trace_paths[0])
    attempts = {a['animation']: a for a in trace['attempts']}
    expected = [dict(kind='archive', archive_frame=0, paired_archive_frame=0, animation=None)]
    for a in trace['attempts']:
        if not a['committed']:
            continue
        for f in range(a['start_frame']+1, a['end_frame']+1):
            if f == a['end_frame']:
                expected.append(dict(kind='optimizer_raw_endpoint', archive_frame=None,
                                     paired_archive_frame=f, animation=a['animation']))
            expected.append(dict(kind='promoted_endpoint' if f == a['end_frame'] else 'archive',
                                 archive_frame=f, paired_archive_frame=f, animation=a['animation']))
    require(len(expected) == len(samples), 'Sample count/physical scope')
    for row, want in zip(samples, expected):
        require({k: row[k] for k in want} == want, 'Raw/promoted/archive schedule mismatch')
    require(result['accepted_archive_frames'] == [r['archive_frame'] for r in expected if r['archive_frame'] is not None],
            'Accepted archive mapping')
    require([i for c in result['chunks'] for i in c['sample_indices']] == list(range(len(samples))), 'Chunk membership')
    chosen = chosen_samples(samples, result['resolutions'])
    violations, reduction_checks, selected_results = [], 0, []
    def equal(actual, want, name):
        nonlocal reduction_checks
        reduction_checks += 1
        actual = actual.detach().double() if torch.is_tensor(actual) else torch.as_tensor(actual, dtype=torch.float64, device='cuda:0')
        want = torch.as_tensor(want, dtype=torch.float64, device=actual.device)
        if actual.shape != want.shape:
            violations.append(dict(check=name, reason='shape', actual=list(actual.shape), expected=list(want.shape)))
        else:
            error = (actual-want).abs()
            if not bool(torch.isfinite(actual).all()) or bool((error > 5e-14+2e-12*want.abs()).any()):
                violations.append(dict(check=name, reason='value', max_abs_error=error.max().item()))
    with torch.inference_mode(), cp.cuda.Device(0), cp.cuda.ExternalStream(torch.cuda.current_stream().cuda_stream):
        with np.load(raw, allow_pickle=False) as z:
            target = torch.as_tensor(z['tgt'].copy(), device='cuda:0')
        require(target.shape == (300000, 3) and target.dtype == torch.float32, 'Target identity/layout')
        # Source spacing is not used here; independently recompute target spacing
        # with the same external KDTree library, bypassing project wrappers.
        td = cp.from_dlpack(target.double())
        spacing = float(cp.median(KDTree(td).query(td, k=2)[0][:, 1]))
        equal(spacing, result['target_spacing_wu'], 'target_native_spacing')
        extent = float(result['extent_wu'])
        extent64 = target.double().square().sum(1).sqrt().max().item()*1.15
        extent_tolerance = 4*torch.finfo(torch.float32).eps*extent64
        require(math.isfinite(extent) and abs(extent-extent64) <= extent_tolerance,
                'Fixed target extent outside FP32 roundoff bound')
        target_masks, target_holes = {}, {}
        with np.load(folder/'target_masks.npz', allow_pickle=False) as z:
            for res in result['resolutions']:
                bits = torch.as_tensor(z[f'target_{res}_bits'].copy(), device='cuda:0')
                target_masks[res] = unpack_rows(bits, res*res).reshape(24, res, res)
        def holes_of(body):
            # Independent bit-mask analysis, no source projection/loss helper.
            return torch.stack([torch.from_dlpack(ndimage.binary_fill_holes(cp.from_dlpack(mask))) & ~mask
                                for mask in body])
        for res, masks in target_masks.items():
            target_holes[res] = holes_of(masks)
            equal(target_holes[res].sum((1, 2)), result['target_hole_pixels'][str(res)], 'target_holes/'+str(res))
            for v, (az, el) in enumerate(result['views']):
                counts, outside = projected_counts(target, res, az, el, extent)
                equal((counts > 0), masks[v], f'target_projection/{res}/{v}')
                require(outside == 0, 'Fixed target extent unexpectedly clips target')
        previous = {}
        selected_cache = {}
        for chunk in result['chunks']:
            with np.load(folder/chunk['path'], allow_pickle=False) as z:
                require(z['sample_indices'].tolist() == chunk['sample_indices'], 'Chunk index payload')
                cover = torch.as_tensor(z['covered_target_bits'].copy(), device='cuda:0')
                covered = unpack_rows(cover, len(target))
                bodies = {res: unpack_rows(torch.as_tensor(z[f'body_{res}_bits'].copy(), device='cuda:0').reshape(-1, (res*res+7)//8),
                                           res*res).reshape(len(covered), 24, res, res) for res in result['resolutions']}
            for local, index in enumerate(chunk['sample_indices']):
                row = samples[index]
                equal(covered[local].sum(), row['target_covered'], f'coverage/{index}')
                require(row['target_particles'] == len(target), 'Coverage denominator')
                for res, stack in bodies.items():
                    body = stack[local]
                    holes = holes_of(body)
                    count, hc = body.sum((1, 2)), holes.sum((1, 2))
                    union = (body | target_masks[res]).sum((1, 2))
                    inter = (body & target_masks[res]).sum((1, 2))
                    values = dict(body_pixels=count, internal_hole_pixels=hc,
                        hole_fraction=hc.double()/(count+hc).clamp_min(1),
                        target_iou=torch.where(union > 0, inter.double()/union.clamp_min(1), 1.),
                        hole_pixels_outside_target_holes=(holes & ~target_holes[res]).sum((1, 2)))
                    if res in previous:
                        values.update(new_hole_pixels=(holes & ~previous[res]).sum((1, 2)),
                                      removed_hole_pixels=(~holes & previous[res]).sum((1, 2)))
                    for name, value in values.items():
                        equal(value, row['views'][str(res)][name], f'packed/{index}/{res}/{name}')
                    previous[res] = holes
                if index in chosen:
                    selected_cache[index] = (covered[local].clone(), {res: b[local].clone() for res, b in bodies.items()})
            print(json.dumps(dict(verified_through_sample=chunk['sample_indices'][-1], violations=len(violations))), flush=True)
        # Recompute source geometry only at the small selected set, after every
        # packed sample's temporal and absolute reductions have been checked.
        for index, reasons in chosen.items():
            row = samples[index]
            if row['kind'] == 'optimizer_raw_endpoint':
                a = attempts[row['animation']]
                p = Path(str(raw).removesuffix('_render_full_dt_iso_nn.npz')+'_cohorts')/a['sidecar']
                require(str(p) in paths and paths[str(p)] == a['sha256'], 'Selected raw endpoint binding')
                with np.load(p, allow_pickle=False) as z:
                    x = torch.as_tensor(z['optimizer_raw_endpoint'].copy(), device='cuda:0')
            else:
                x = read_frame(raw, row['archive_frame'])
            require(x.shape == target.shape and bool(torch.isfinite(x).all()), 'Selected source state')
            cv, saved = selected_cache[index]
            mismatches = {}
            for res in result['resolutions']:
                bad = 0
                outside_counts = []
                for v, (az, el) in enumerate(result['views']):
                    counts, outside = projected_counts(x, res, az, el, extent)
                    mask = counts > 0
                    bad += int((mask != saved[res][v]).sum().item())
                    equal(mask, saved[res][v], f'selected_projection/{index}/{res}/{v}')
                    outside_counts.append(outside)
                equal(outside_counts, row['views'][str(res)]['projected_outside_centers'], f'clipping/{index}/{res}')
                mismatches[str(res)] = bad
            equal((x.abs() > extent).any(1).sum(), row['source_outside_extent_box'], f'box_exclusion/{index}')
            ids = target_ids(cv)
            distances = []
            xd = x.double()
            for start in range(0, len(ids), 4):
                q = target[ids[start:start+4]].double()
                distances.append((q[:, None, :]-xd[None, :, :]).square().sum(-1).min(1).values.sqrt())
            distance = torch.cat(distances)
            brute = distance <= 2*spacing
            equal(brute, cv[ids], f'selected_brute_coverage/{index}')
            selected_results.append(dict(sample_index=index, sample={k: row[k] for k in ('kind','archive_frame','paired_archive_frame','animation')},
                reasons=reasons, projection_mask_mismatches=mismatches, target_ids=ids.tolist(),
                target_nearest_distance_wu=distance.tolist(), covered_by_brute=brute.tolist(),
                stored_covered=cv[ids].tolist(), minimum_cutoff_margin_wu=(distance-2*spacing).abs().min().item()))
        torch.cuda.synchronize()
    require(all(identity(p) == before[p] for p in paths), 'Input/code/output identity changed')
    report = dict(passed=not violations, violations=violations, no_mpm_forward=True, no_renderer=True,
        created_utc=datetime.now(timezone.utc).isoformat(), result_sha256=result_sha,
        bindings_full_hash_checked=paths, stable_file_identities=before,
        discretization=d, arm=result['arm'], total_labelled_samples=len(samples),
        reduction_checks=reduction_checks, selected_source_recomputations=selected_results,
        coverage_counts_reduced_from_all_stored_bits=True,
        all_stored_body_holes_iou_and_transitions_reduced=True,
        all_target_nearest_distances_recomputed=False,
        source_outside_target_support_recomputed=False,
        independent_target_spacing='Direct CuPy14 device KDTree, same external library; no project wrapper',
        target_extent_check=dict(recorded_wu=extent, independent_fp64_wu=extent64,
                                 fp32_roundoff_bound_wu=extent_tolerance),
        hole_fill_scope='Packed body masks re-filled on device with CuPy ndimage, the same external fill library',
        source_projection_scope='All300k source points,24 views,128/256 masks at selected samples only',
        brute_coverage_scope='Up to128 deterministic target IDs per selected sample, all300k source distances in float64',
        natural_rest_established=False, watertightness_established=False, four_k_appearance_established=False,
        torch_peak_bytes=torch.cuda.max_memory_allocated(), main_elapsed_s=time.perf_counter()-started)
    with output.open('x') as stream:
        json.dump(report, stream, indent=2, allow_nan=False)
    print(json.dumps(dict(passed=report['passed'], samples=len(samples), selected_samples=len(chosen),
                          reduction_checks=reduction_checks, violations=len(violations))))
    require(report['passed'], 'Independent numerical check failed; receipt preserved')


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--analysis', type=Path, required=True)
    parser.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    main(args.analysis, args.out)
