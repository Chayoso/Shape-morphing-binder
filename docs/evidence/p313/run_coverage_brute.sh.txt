#!/bin/bash
set -euo pipefail
base=/data/relcfd/chayo/physmorph_v2
code="$base/work/p303/code_coverage_paths1"
audit="$base/work/p303/coverage_brute_audit1"
out="$base/work/p303/coverage_brute1"
export PHYSMORPH_RUN_REPO="$code"
source "$code/scripts/ops/hyde06_env.sh"
source "$code/scripts/ops/gpu_env.sh"
export CUDA_VISIBLE_DEVICES=1
exec 9>"$base/maintenance/gpu_launch.lock"
flock -x 9
now=$(date +%s)
last=0
for marker in "$base"/repro/current_pair/launch_marker_*.txt "$base/maintenance/last_gpu_launch_epoch"; do
    if test -f "$marker"; then value=$(cat "$marker"); (( value <= last )) || last=$value; fi
done
if (( now-last < 50 )); then
    delay=$((50-now+last))
    (( delay <= 50 )) || exit 75
    sleep "$delay"
    now=$(date +%s)
fi
for suffix in '' .start .log; do test ! -e "$out$suffix"; done
used=$(du -sb "$base" | cut -f1)
(( used+10000000 < 100000000000 )) || exit 76
test -z "$(nvidia-smi -i 1 --query-compute-apps=pid --format=csv,noheader)"
(set -o noclobber; printf '%s\n' "$now" > "$out.start")
set -o noclobber
exec > "$out.log" 2>&1
set +o noclobber
printf '%s\n' "$now" > "$base/maintenance/last_gpu_launch_epoch"
flock -u 9
cd "$code"
set +e
"$PY" scripts/ops/cuda_python.py "$audit/coverage_brute_audit.py" --base "$base" --code "$code" --source "$base/work/p303/coverage_paths1" --out "$out"
rc=$?
printf '%s\n' "$rc" > "$out.exit"
exit "$rc"
